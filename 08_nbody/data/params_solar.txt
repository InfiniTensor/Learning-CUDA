# 太阳系参数档（配合 --preset solar_system）
#
# 单位与初值文件一致：G=1，长度 AU，质量 M_sun，时间 1/k = 58.1324 天
# （k = 0.01720209895 rad/day）。此时 GM_sun = 1，a=1 AU 的周期恰为 2*pi，
# 1 儒略年 = 6.283185 个时间单位。
#
# 为什么不能沿用星团档（params_cluster.txt: dt=2e-3, softening=0.02）：
# 太阳系的质量比横跨 6 个数量级（太阳 1.0 vs 水星 1.66e-7），最内侧的水星
# a=0.387 AU，而 softening 是全局量。实测（build_ic/solar_dt.py，独立 python
# leapfrog，12 年积分）：
#
#   eps(AU)   eps/a_merc    max|dE/E|    水星 a 的相对漂移
#     2e-2     5.2e-02      2.06e-09        2.2e-03      <- 星团档，废掉了水星
#     1e-3     2.6e-03      2.22e-09        1.3e-06
#     1e-4     2.6e-04      2.22e-09        8.4e-07      <- 本档
#     1e-5     2.6e-05      2.22e-09        8.6e-07      <- 再小已无收益
#
# 注意 softening 几乎不影响能量守恒（dE/E 四档都是 2.2e-9）——因为软化后的
# 系统仍有严格守恒的能量，只是它对应的不是真正的牛顿引力。所以**只看能量
# 守恒会误判 eps=0.02 是安全的**，必须看轨道要素才暴露出 2600 倍的差距。
#
# dt 的选择由水星定（周期 88 天 = 1.513 时间单位）。同一测量给出：
#
#      dt      步数/12yr   max|dE/E|
#    5e-3        15080      2.14e-07
#    2e-3        37699      3.52e-08
#    1e-3        75398      8.81e-09
#    5e-4       150796      2.22e-09     <- 本档
#    2e-4       376991      3.44e-10
#
# 误差严格按 dt^2 走（dt 降 5 倍，误差降 24.3 倍 ≈ 5^2），符合 leapfrog 的
# 二阶精度。取 5e-4 即水星每周期约 3000 步，dE/E ~ 2e-9。
#
# num_steps = 150796 ≈ 12 儒略年，刚好覆盖木星一整圈（11.862 yr），
# 动画里能看到最外侧行星也走完一个完整轨道。
# record_interval = 100 -> 1508 帧，画轨迹足够平滑。
#
# 提醒：这套初值来自 J2000 平均轨道要素，位置与 JPL 历表差 ~1e-4 AU 量级。
# 用来验证"能否长期维持稳定轨道"完全够用，但它不是星历预报，也给不出水星
# 近日点进动里那 43 arcsec/世纪 的广义相对论部分。
dt = 5e-4
num_steps = 150796
record_interval = 100
G = 1.0
softening = 1e-4
integrator = "leapfrog"
