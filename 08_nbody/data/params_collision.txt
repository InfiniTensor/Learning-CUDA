# 两 Plummer 球对撞 / 并合的参数（cluster_collision 预设）
#
# 为什么需要独立的 params 文件、不能套用 params_cluster.txt：
# 星团档假设系统始终接近平衡、粒子间距大致均匀，softening=0.02 够用；
# 而并合过程中两球互相穿透，会出现真正的近距离交会——
# softening 太大会把并合时的核心塌缩抹平（看不到现象），
# 太小则加速度峰值撑不住 dt（积分器发散）。所以两个值都要实测定稿。
#
# ---------------------------------------------------------------------------
# dt 的取值是实测出来的，不是拍脑袋
# ---------------------------------------------------------------------------
# 判据：跑到同一物理时刻 t=20（已穿过第一次近星点 t≈13.4），比较能量漂移。
# 在 MX330 上用 N=4096、softening=0.005 实测：
#
#   dt      步数     |dE/E0|      结论
#   4e-3    5000     1.469e-05    ← 明显退化，已出收敛区
#   2e-3    10000    5.642e-07    ← 选这个：已在收敛区内
#   1e-3    20000    2.834e-07
#   5e-4    40000    2.165e-07
#
# 从 4e-3 到 2e-3 误差降 26 倍，而从 2e-3 再降到 5e-4 只再降 2.6 倍——
# 说明 2e-3 已经进入收敛区（误差由其他因素主导，不再由 dt 主导），
# 继续减小 dt 只是多花时间换不到精度。拐点在 4e-3，所以 2e-3 留了 2 倍余量。
dt = 2e-3

# ---------------------------------------------------------------------------
# 步数要覆盖真正的并合，而不只是"两球靠近"
# ---------------------------------------------------------------------------
# 默认碰撞参数（d=8a, v_rel=0.7 v_esc, b=1.5a）下的解析轨道要素：
#   a_orb = 7.84   e = 0.982   r_peri = 0.139   r_apo = 15.5
#   轨道周期 T = 138（G=M=a=1 单位制）
#   从 d=8 到第一次近星点约 t = 13.4
#
# 这里最容易犯的错是**步数给太少**：跑到 t=3 时两球只从 7.94 靠近到 2.12，
# 看起来"在接近"但根本没并合，动画会给出"什么都没发生"的错误印象。
# t=300 覆盖两个多轨道周期，足够看到"穿过 → 分开 → 再穿过 → 并合"的完整过程。
num_steps = 150000

# 150000 / 750 = 200 个记录点。N=4096 时轨迹文件 4096*201*12 ≈ 9.9 MB，
# 对 matplotlib 动画来说 200 帧也正合适。
record_interval = 750

G = 1.0

# ---------------------------------------------------------------------------
# softening：比星团档小 4 倍
# ---------------------------------------------------------------------------
# 星团档用 0.02（≈ 平均粒子间距量级，系统始终接近平衡）。
# 并合要小一些：r_peri = 0.139，softening=0.02 已是近星点距离的 14%，
# 会显著改变最接近时刻的动力学。取 0.005（3.6%）在"不抹平核心塌缩"与
# "dt 撑得住加速度峰值"之间；上面的 dt 标定就是在这个 softening 下做的，
# 两者是配套的——改 softening 必须重新标定 dt。
softening = 0.005

integrator = "leapfrog"
format = "binary"
