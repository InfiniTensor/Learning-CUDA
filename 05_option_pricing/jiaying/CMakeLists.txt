cmake_minimum_required(VERSION 3.24)
project(option_pricing LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(OPTION_ENABLE_CUDA "Build CUDA backend" ON)
add_executable(option_pricing src/main.cpp src/io.cpp src/cpu.cpp)
target_include_directories(option_pricing PRIVATE include)
target_compile_options(option_pricing PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-O3;-Wall;-Wextra;-Wpedantic>)
if(OPTION_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES AND NOT DEFINED ENV{CUDAARCHS})
    set(CMAKE_CUDA_ARCHITECTURES 75 86 89)
  endif()
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  target_sources(option_pricing PRIVATE src/cuda.cu)
  target_compile_definitions(option_pricing PRIVATE OPTION_HAS_CUDA=1)
  target_compile_options(option_pricing PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;-lineinfo>)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(NOT OPTION_ENABLE_CUDA)
  set(VALIDATION_ARGS --cpu-only)
endif()
add_test(NAME pricing_validation COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_CURRENT_SOURCE_DIR}/tests/validate.py --exe $<TARGET_FILE:option_pricing>
  --output ${CMAKE_CURRENT_BINARY_DIR}/validation ${VALIDATION_ARGS})
set_tests_properties(pricing_validation PROPERTIES TIMEOUT 300)
