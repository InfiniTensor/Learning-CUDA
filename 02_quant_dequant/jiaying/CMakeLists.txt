cmake_minimum_required(VERSION 3.20)
project(low_precision LANGUAGES CXX)
option(LP_ENABLE_CUDA "Build the portable CUDA backend" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
add_library(lp_core src/core.cpp src/io.cpp)
target_include_directories(lp_core PUBLIC include)
target_compile_options(lp_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
if(LP_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 60 75 89)
  endif()
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  add_library(lp_gpu src/cuda_backend.cu)
  target_include_directories(lp_gpu PUBLIC include)
  target_compile_options(lp_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-lineinfo>)
  target_compile_definitions(lp_core PUBLIC LP_ENABLE_CUDA=1)
endif()
add_executable(low_precision src/main.cpp)
target_link_libraries(low_precision PRIVATE lp_core)
if(LP_ENABLE_CUDA)
  target_link_libraries(low_precision PRIVATE lp_gpu)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(codec_tests tests/codec_tests.cpp)
  target_link_libraries(codec_tests PRIVATE lp_core)
  add_test(NAME codecs COMMAND codec_tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME independent_reference COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:low_precision>
      --backend cpu)
    if(LP_ENABLE_CUDA)
      add_test(NAME cuda_reference COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:low_precision>
        --backend cuda)
    endif()
  endif()
endif()
