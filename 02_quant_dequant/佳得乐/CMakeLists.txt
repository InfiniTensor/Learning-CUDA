cmake_minimum_required(VERSION 3.16)
project(low_precision LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ENABLE_CUDA "Build software CUDA backend" ON)
set(CUDA_ARCH "86" CACHE STRING "CUDA SM architecture")
add_library(lp_cpu src/cpu.cpp)
target_include_directories(lp_cpu PUBLIC include)
target_compile_options(lp_cpu PRIVATE -O3 -Wall -Wextra -ffp-contract=off)
add_executable(scalar_tests tests/scalar.cpp)
target_link_libraries(scalar_tests PRIVATE lp_cpu)
target_compile_options(scalar_tests PRIVATE -O3)
enable_testing()
add_test(NAME scalar COMMAND scalar_tests)
target_sources(lp_cpu PRIVATE src/io.cpp)
if(ENABLE_CUDA)
 enable_language(CUDA)
 add_library(lp_gpu src/gpu.cu)
 target_include_directories(lp_gpu PUBLIC include)
 target_compile_options(lp_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-std=c++17;-O3;--fmad=false;-arch=sm_${CUDA_ARCH};-lineinfo>)
 target_link_libraries(lp_gpu PUBLIC lp_cpu)
else()
 add_library(lp_gpu src/gpu_stub.cpp)
 target_link_libraries(lp_gpu PUBLIC lp_cpu)
endif()
add_executable(lp src/main.cpp)
target_link_libraries(lp PRIVATE lp_gpu)
target_compile_options(lp PRIVATE -O3 -ffp-contract=off)
add_executable(integration_tests tests/integration.cpp)
target_link_libraries(integration_tests PRIVATE lp_gpu)
add_test(NAME integration_cpu COMMAND integration_tests)
if(ENABLE_CUDA)
 add_test(NAME integration_gpu COMMAND integration_tests gpu)
endif()
add_executable(quality tests/quality.cpp)
target_link_libraries(quality PRIVATE lp_cpu)
